import { getStorageData, saveStorageData } from './storage.js'; // 初始化個人自購模組 export function initPersonalModule() { const container = document.getElementById('personal-items-container'); const searchInput = document.getElementById('personal-search'); const filterStatus = document.getElementById('personal-filter-status'); const filterPay = document.getElementById('personal-filter-pay'); const btnOpenModal = document.getElementById('btn-open-personal-modal'); const btnCancelModal = document.getElementById('btn-cancel-personal'); const modal = document.getElementById('modal-personal'); const form = document.getElementById('form-personal'); const payMethodSelect = document.getElementById('personal-paymethod-select'); const depositContainer = document.getElementById('field-deposit-container'); const modalTitle = document.getElementById('modal-personal-title'); const btnSubmit = document.getElementById('btn-submit-personal'); // 控制訂金欄位顯示 payMethodSelect.addEventListener('change', () => { if (payMethodSelect.value === '訂金+尾款') { depositContainer.classList.remove('hidden'); } else { depositContainer.classList.add('hidden'); } }); // 打開新增彈窗 btnOpenModal.addEventListener('click', () => { form.reset(); document.getElementById('personal-item-id').value = ''; modalTitle.textContent = '新增個人自購商品'; btnSubmit.textContent = '儲存'; depositContainer.classList.add('hidden'); modal.classList.remove('hidden'); }); // 關閉彈窗 btnCancelModal.addEventListener('click', () => { modal.classList.add('hidden'); }); // 搜尋與篩選監聽 searchInput.addEventListener('input', renderCards); filterStatus.addEventListener('change', renderCards); filterPay.addEventListener('change', renderCards); // 表單提交（新增或編輯） form.addEventListener('submit', (e) => { e.preventDefault(); const formData = new FormData(form); const data = getStorageData(); const itemId = formData.get('id'); const itemData = { name: formData.get('name'), imageUrl: formData.get('imageUrl') || '', quantity: Number(formData.get('quantity')) || 1, price: Number(formData.get('price')) || 0, channel: formData.get('channel') || '', payMethod: formData.get('payMethod'), deposit: formData.get('payMethod') === '訂金+尾款' ? (Number(formData.get('deposit')) || 0) : 0, orderDate: formData.get('orderDate') || '', arrivalMonth: formData.get('arrivalMonth') || '', status: formData.get('status') || '已喊單', notes: formData.get('notes') || '' }; if (itemId) { // 編輯更新 const index = data.personalItems.findIndex(i => i.id === itemId); if (index !== -1) { data.personalItems[index] = { ...data.personalItems[index], ...itemData }; } } else { // 全新新增 itemData.id = 'p_' + Date.now(); itemData.createdAt = new Date().toISOString(); if (!data.personalItems) data.personalItems = []; data.personalItems.push(itemData); } saveStorageData(data); modal.classList.add('hidden'); renderCards(); window.dispatchEvent(new CustomEvent('data-updated')); }); // 渲染所有卡片 function renderCards() { const data = getStorageData(); let list = data.personalItems || []; const keyword = searchInput.value.trim().toLowerCase(); const statusVal = filterStatus.value; const payVal = filterPay.value; list = list.filter(item => { const matchKeyword = !keyword || (item.name && item.name.toLowerCase().includes(keyword)) || (item.channel && item.channel.toLowerCase().includes(keyword)); const matchStatus = !statusVal || item.status === statusVal; const matchPay = !payVal || item.payMethod === payVal; return matchKeyword && matchStatus && matchPay; }); if (list.length === 0) { container.innerHTML = `
目前無相關自購項目
`; return; } container.innerHTML = list.map(item => { const totalPrice = item.price * item.quantity; const deposit = item.deposit || 0; const balance = Math.max(0, totalPrice - deposit); return `

${item.name}

${item.channel || '無管道'} ${item.payMethod} ${item.status}
單價 × 數量 NT$ ${item.price.toLocaleString()} × ${item.quantity}
總價 NT$ ${totalPrice.toLocaleString()}
${item.payMethod === '訂金+尾款' ? `
已付訂金 NT$ ${deposit.toLocaleString()}
剩餘尾款 NT$ ${balance.toLocaleString()}
` : ''}
下單 / 抵台 ${item.orderDate || '--'} / ${item.arrivalMonth || '--'}
更新物流進度
`; }).join(''); } // 掛載編輯函式到 window 以供 onclick 呼叫 window.editPersonalItem = function(id) { const data = getStorageData(); const item = (data.personalItems || []).find(i => i.id === id); if (!item) return; // 填入既有數據 document.getElementById('personal-item-id').value = item.id; document.getElementById('personal-name').value = item.name || ''; document.getElementById('personal-imageUrl').value = item.imageUrl || ''; document.getElementById('personal-quantity').value = item.quantity || 1; document.getElementById('personal-price').value = item.price || 0; document.getElementById('personal-channel').value = item.channel || ''; document.getElementById('personal-paymethod-select').value = item.payMethod || '取貨付款'; document.getElementById('personal-deposit').value = item.deposit || 0; document.getElementById('personal-orderDate').value = item.orderDate || ''; document.getElementById('personal-arrivalMonth').value = item.arrivalMonth || ''; document.getElementById('personal-status').value = item.status || '已喊單'; document.getElementById('personal-notes').value = item.notes || ''; // 調整訂金容器顯示 if (item.payMethod === '訂金+尾款') { depositContainer.classList.remove('hidden'); } else { depositContainer.classList.add('hidden'); } modalTitle.textContent = '編輯自購商品'; btnSubmit.textContent = '儲存變更'; modal.classList.remove('hidden'); }; // 掛載狀態更新函式到 window window.updatePersonalStatus = function(id, newStatus) { const data = getStorageData(); const item = (data.personalItems || []).find(i => i.id === id); if (item) { item.status = newStatus; saveStorageData(data); renderCards(); window.dispatchEvent(new CustomEvent('data-updated')); } }; // 初始渲染 renderCards(); }
